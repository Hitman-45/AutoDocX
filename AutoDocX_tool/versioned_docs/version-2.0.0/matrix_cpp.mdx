      vector&lt;vector&lt;int&gt;&gt; transposeMatrix(const vector&lt;vector&lt;int&gt;&gt;& mat) {
    int rows = mat.size();
    int cols = mat[0].size();
    vector&lt;vector&lt;int&gt;&gt; trans(cols, vector&lt;int&gt;(rows));

    for (int i = 0; i &lt; rows; ++i)
